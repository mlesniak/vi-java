# vi-Editor Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ein lauffähiger, minimaler vi-Klon in plain Java (Normal/Insert/Command-Mode, Undo, Laden/Speichern) unter 1024 Zeilen Produktivcode.

**Architecture:** Fünf Klassen im Paket `com.mlesniak.vi`. `Buffer` ist das Textmodell mit Undo-Snapshots, `Editor` die reine Zustandsmaschine (kein IO), `Screen` rendert den Zustand in einen ANSI-Frame-String, `Terminal` kapselt Raw-Mode über `stty` und Tastatur-Bytes, `Vi` verdrahtet alles in der Hauptschleife. Nur `Terminal` und `Vi` berühren stdin/stdout, alles andere ist mit JUnit testbar.

**Tech Stack:** Java 25, Maven 3.9, JUnit 5 (nur Test-Scope). Keine Laufzeit-Dependencies.

**Spec:** `docs/superpowers/specs/2026-09-08-vi-editor-design.md`

## Global Constraints

- Java 25 (`maven.compiler.release` = 25), Maven, keine Laufzeit-Dependencies. JUnit 5 nur mit `<scope>test</scope>`.
- Produktivcode (`src/main/java/com/mlesniak/vi/*.java`) zusammen unter 1024 Zeilen. Messung: `cat src/main/java/com/mlesniak/vi/*.java | wc -l`.
- Formatierung: 4 Leerzeichen Einrückung, max. 100 Zeichen pro Zeile, geschweifte Klammern immer (auch bei einzeiligen `if`/`while`), ein Statement pro Zeile. Kein Golfen, um Zeilen zu sparen.
- Switch-Expressions und `var` sind erlaubt, wo sie lesbarer sind.
- Nur Linux/macOS (Raw-Mode via `stty` und `/dev/tty`).
- Code, Kommentare und Commit-Messages auf Englisch. Jede Klasse bekommt einen einzeiligen Javadoc-Kommentar, sonst nur Kommentare, wo der Code nicht selbsterklärend ist.
- Alle Tests laufen mit `mvn -q test`. Vor jedem Commit müssen sie grün sein.
- Tastencodes sind `int`: `Terminal.ESC = 27`, `Terminal.ENTER = 13`, `Terminal.BACKSPACE = 127`, `Terminal.UP = 1000`, `Terminal.DOWN = 1001`, `Terminal.RIGHT = 1002`, `Terminal.LEFT = 1003`. Zusätzlich gilt `8` als Backspace.

```java
package com.mlesniak.vi;

import java.util.ArrayList;
import java.util.List;

/** Text model: a list of lines plus file name and dirty flag. Never has zero lines. */
public class Buffer {
    private List<String> lines;
    private String name;
    private boolean dirty;

    public Buffer() {
        this(List.of(""));
    }

    public Buffer(List<String> initial) {
        lines = new ArrayList<>(initial);
        if (lines.isEmpty()) {
            lines.add("");
        }
    }

    public String line(int row) {
        return lines.get(row);
    }

    public int size() {
        return lines.size();
    }

    public String name() {
        return name;
    }

    public void name(String name) {
        this.name = name;
    }

    public boolean dirty() {
        return dirty;
    }

    public void insert(int row, int col, char c) {
        String s = lines.get(row);
        lines.set(row, s.substring(0, col) + c + s.substring(col));
        dirty = true;
    }

    public void delete(int row, int col) {
        String s = lines.get(row);
        lines.set(row, s.substring(0, col) + s.substring(col + 1));
        dirty = true;
    }

    public void split(int row, int col) {
        String s = lines.get(row);
        lines.set(row, s.substring(0, col));
        lines.add(row + 1, s.substring(col));
        dirty = true;
    }

    public void join(int row) {
        lines.set(row, lines.get(row) + lines.remove(row + 1));
        dirty = true;
    }

    public void insertLine(int row, String s) {
        lines.add(row, s);
        dirty = true;
    }

    public String deleteLine(int row) {
        String s = lines.remove(row);
        if (lines.isEmpty()) {
            lines.add("");
        }
        dirty = true;
        return s;
    }
}
```

```java
package com.mlesniak.vi;

/** Editor state machine: modes, cursor, scrolling and key dispatch. Does no terminal IO. */
public class Editor {
    public enum Mode {
        NORMAL, INSERT, COMMAND
    }

    private Buffer buf;
    private Mode mode = Mode.NORMAL;
    private int row;
    private int col;
    private int scroll;
    private final int rows;
    private final int cols;
    private final StringBuilder cmd = new StringBuilder();
    private String status = "";
    private boolean running = true;

    public Editor(Buffer buf, int rows, int cols) {
        this.buf = buf;
        this.rows = rows;
        this.cols = cols;
    }

    public void handleKey(int key) {
        status = "";
        switch (mode) {
            case NORMAL -> normal(key);
            case INSERT -> {
            }
            case COMMAND -> {
            }
        }
        clampCursor();
        adjustScroll();
    }

    private void normal(int key) {
        switch (key) {
            case 'h', Terminal.LEFT -> col--;
            case 'l', Terminal.RIGHT -> col++;
            case 'j', Terminal.DOWN -> row++;
            case 'k', Terminal.UP -> row--;
            case '0' -> col = 0;
            case '$' -> col = buf.line(row).length();
            default -> {
            }
        }
    }

    /** Keeps row inside the buffer and col on a character (or one past the end in INSERT). */
    private void clampCursor() {
        row = Math.max(0, Math.min(row, buf.size() - 1));
        int max = buf.line(row).length();
        if (mode != Mode.INSERT) {
            max = Math.max(0, max - 1);
        }
        col = Math.max(0, Math.min(col, max));
    }

    private void adjustScroll() {
        int textRows = rows - 1;
        if (row < scroll) {
            scroll = row;
        } else if (row >= scroll + textRows) {
            scroll = row - textRows + 1;
        }
    }

    public Mode mode() {
        return mode;
    }

    public int row() {
        return row;
    }

    public int col() {
        return col;
    }

    public int scroll() {
        return scroll;
    }

    public int rows() {
        return rows;
    }

    public int cols() {
        return cols;
    }

    public String status() {
        return status;
    }

    public String cmd() {
        return cmd.toString();
    }

    public Buffer buffer() {
        return buf;
    }

    public boolean running() {
        return running;
    }
}
```

```java
    private void normal(int key) {
        if (pending != 0) {
            pendingKey(key);
            return;
        }
        switch (key) {
            case 'h', Terminal.LEFT -> col--;
            case 'l', Terminal.RIGHT -> col++;
            case 'j', Terminal.DOWN -> row++;
            case 'k', Terminal.UP -> row--;
            case '0' -> col = 0;
            case '$' -> col = buf.line(row).length();
            case 'w' -> wordForward();
            case 'b' -> wordBackward();
            case 'G' -> row = buf.size() - 1;
            case 'g' -> pending = key;
            default -> {
            }
        }
    }

    /** Second key of a two-key command (gg). Any other combination is dropped. */
    private void pendingKey(int key) {
        int op = pending;
        pending = 0;
        if (op == 'g' && key == 'g') {
            row = 0;
        }
    }

    private void wordForward() {
        int r = row;
        int c = col;
        String s = buf.line(r);
        while (c < s.length() && !Character.isWhitespace(s.charAt(c))) {
            c++;
        }
        while (true) {
            while (c < s.length() && Character.isWhitespace(s.charAt(c))) {
                c++;
            }
            if (c < s.length() || r == buf.size() - 1) {
                break;
            }
            r++;
            c = 0;
            s = buf.line(r);
        }
        row = r;
        col = c;
    }

    private void wordBackward() {
        int r = row;
        int c = col;
        while (true) {
            String s = buf.line(r);
            while (c > 0 && Character.isWhitespace(s.charAt(c - 1))) {
                c--;
            }
            if (c > 0 || r == 0) {
                break;
            }
            r--;
            c = buf.line(r).length();
        }
        String s = buf.line(r);
        while (c > 0 && !Character.isWhitespace(s.charAt(c - 1))) {
            c--;
        }
        row = r;
        col = c;
    }
```

```java
    private void enterInsert() {
        buf.snapshot();

    }

    private void insert(int key) {
        switch (key) {
            case Terminal.ESC -> {
                mode = Mode.NORMAL;
                col--;

                buf.split(row, col);
                row++;
                col = 0;

            buf.delete(row, col);
        } else if (row > 0) {

            buf.join(row - 1);
            row--;
        }
```

```java
    private void command(int key) {
        switch (key) {
            case Terminal.ESC -> mode = Mode.NORMAL;
            case Terminal.ENTER -> {
                mode = Mode.NORMAL;
                execute(cmd.toString().trim());
            }
            case Terminal.BACKSPACE, 8 -> {
                if (cmd.isEmpty()) {
                    mode = Mode.NORMAL;
                } else {
                    cmd.setLength(cmd.length() - 1);
                }
            }
            default -> {
                if (key >= 32 && key <= 126) {
                    cmd.append((char) key);
                }
            }
        }
    }

    private void execute(String line) {
        if (line.isEmpty()) {
            return;
        }
        String[] parts = line.split("\\s+", 2);
        String arg = parts.length > 1 ? parts[1] : null;
        try {
            switch (parts[0]) {
                case "w" -> write(arg);
                case "q" -> quit(false);
                case "q!" -> quit(true);
                case "wq" -> {
                    write(arg);
                    quit(false);
                }
                case "e" -> edit(arg);
                default -> status = "Not an editor command: " + line;
            }
        } catch (IOException e) {
            status = e.getMessage();
        }
    }

    private void write(String name) throws IOException {
        if (name != null) {
            buf.name(name);
        }
        buf.save();
        status = "\"" + buf.name() + "\" written";
    }

    private void quit(boolean force) {
        if (buf.dirty() && !force) {
            status = "No write since last change (add ! to override)";
        } else {
            running = false;
        }
    }

    private void edit(String name) throws IOException {
        if (name == null) {
            throw new IOException("No file name");
        }
        if (buf.dirty()) {
            status = "No write since last change (add ! to override)";
            return;
        }
        buf = Buffer.load(name);
        row = 0;
        col = 0;
        scroll = 0;
    }
```

```java
package com.mlesniak.vi;

/** Renders the editor state into one complete ANSI frame string. */
public final class Screen {
    private static final String CSI = "\033[";

    private Screen() {
    }

    public static String render(Editor e) {
        Buffer buf = e.buffer();
        StringBuilder sb = new StringBuilder(CSI + "?25l" + CSI + "H");
        for (int i = 0; i < e.rows() - 1; i++) {
            int r = e.scroll() + i;
            String line = r < buf.size() ? buf.line(r) : "~";
            sb.append(line, 0, Math.min(line.length(), e.cols()));
            sb.append(CSI).append("K\r\n");
        }
        String bottom = bottomLine(e);
        sb.append(bottom, 0, Math.min(bottom.length(), e.cols() - 1));
        sb.append(CSI).append('K');
        int cy;
        int cx;
        if (e.mode() == Editor.Mode.COMMAND) {
            cy = e.rows();
            cx = e.cmd().length() + 2;
        } else {
            cy = e.row() - e.scroll() + 1;
            cx = Math.min(e.col(), e.cols() - 1) + 1;
        }
        sb.append(CSI).append(cy).append(';').append(cx).append('H');
        sb.append(CSI).append("?25h");
        return sb.toString();
    }

    private static String bottomLine(Editor e) {
        if (e.mode() == Editor.Mode.COMMAND) {
            return ":" + e.cmd();
        }
        if (!e.status().isEmpty()) {
            return e.status();
        }
        Buffer buf = e.buffer();
        String name = buf.name() == null ? "[No Name]" : buf.name();
        String flag = buf.dirty() ? " [+]" : "";
        String mode = e.mode() == Editor.Mode.INSERT ? "  -- INSERT --" : "";
        return name + flag + mode + "  " + (e.row() + 1) + "," + (e.col() + 1);
    }
}
```

```java
package com.mlesniak.vi;

import java.io.File;
import java.io.IOException;
import java.nio.charset.StandardCharsets;

/** Raw-mode terminal access via stty and ANSI escape sequences. Linux/macOS only. */
public class Terminal {
    public static final int ESC = 27;
    public static final int ENTER = 13;
    public static final int BACKSPACE = 127;
    public static final int UP = 1000;
    public static final int DOWN = 1001;
    public static final int RIGHT = 1002;
    public static final int LEFT = 1003;

    private String saved;
    private int pushback = -1;

    public void enableRaw() throws IOException {
        saved = stty("-g").trim();
        stty("raw", "-echo");
        write("\033[?1049h");
    }

    public void restore() {
        write("\033[?1049l");
        if (saved == null) {
            return;
        }
        try {
            stty(saved);
        } catch (IOException e) {
            System.err.println("vi: cannot restore terminal: " + e.getMessage());
        }
    }

    /** Returns {rows, cols}; falls back to 24x80 if stty is unavailable. */
    public int[] size() {
        try {
            String[] parts = stty("size").trim().split(" ");
            return new int[] {Integer.parseInt(parts[0]), Integer.parseInt(parts[1])};
        } catch (IOException | RuntimeException e) {
            return new int[] {24, 80};
        }
    }

    /** Reads one key; arrow escape sequences are folded into UP/DOWN/RIGHT/LEFT. EOF is -1. */
    public int readKey() throws IOException {
        int key = next();
        if (key != ESC || System.in.available() == 0) {
            return key;
        }
        int second = next();
        if (second != '[') {
            pushback = second;
            return ESC;
        }
        return switch (next()) {
            case 'A' -> UP;
            case 'B' -> DOWN;
            case 'C' -> RIGHT;
            case 'D' -> LEFT;
            default -> ESC;
        };
    }

    public void write(String s) {
        System.out.print(s);
        System.out.flush();
    }

    private int next() throws IOException {
        if (pushback >= 0) {
            int key = pushback;
            pushback = -1;
            return key;
        }
        return System.in.read();
    }

    private static String stty(String... args) throws IOException {
        String[] cmd = new String[args.length + 1];
        cmd[0] = "stty";
        System.arraycopy(args, 0, cmd, 1, args.length);
        Process p = new ProcessBuilder(cmd).redirectInput(new File("/dev/tty")).start();
        String out = new String(p.getInputStream().readAllBytes(), StandardCharsets.UTF_8);
        try {
            if (p.waitFor() != 0) {
                throw new IOException("stty " + String.join(" ", args) + " failed");
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new IOException("interrupted while running stty", e);
        }
        return out;
    }
}
```

```java
package com.mlesniak.vi;

import java.io.IOException;

/** Entry point: wires terminal, buffer, editor and screen into the main loop. */
public final class Vi {
    private Vi() {
    }

    public static void main(String[] args) {
        Buffer buf;
        try {
            buf = args.length > 0 ? Buffer.load(args[0]) : new Buffer();
        } catch (IOException e) {
            System.err.println("vi: " + e.getMessage());
            System.exit(1);
            return;
        }
        Terminal term = new Terminal();
        try {
            term.enableRaw();
            int[] size = term.size();
            Editor editor = new Editor(buf, size[0], size[1]);
            while (editor.running()) {
                term.write(Screen.render(editor));
                int key = term.readKey();
                if (key < 0) {
                    break;
                }
                editor.handleKey(key);
            }
        } catch (IOException e) {
            System.err.println("vi: " + e.getMessage());
        } finally {
            term.restore();
        }
    }
}
```
